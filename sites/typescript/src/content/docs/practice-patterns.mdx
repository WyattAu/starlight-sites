## Exercise 1: Singleton

**Problem:**
Implement a thread-safe singleton in TypeScript using a private constructor and static instance.

**Hint:** Use a static `getInstance` method and a private constructor.

```typescript
class Database \{
  private static instance: Database;
  private connected: boolean = false;

  private constructor() \{\}

  static getInstance(): Database \{
    if (!Database.instance) \{
      Database.instance = new Database();
    \}
    return Database.instance;
  \}

  connect(): void \{
    if (!this.connected) \{
      this.connected = true;
      console.log("database connected");
    \}
  \}

  isConnected(): boolean \{
    return this.connected;
  \}
\}

const db1 = Database.getInstance();
const db2 = Database.getInstance();
console.log(db1 === db2); // true
```

## Exercise 2: Observer

**Problem:**
Implement an Observer pattern with typed subscriptions and notification.

**Hint:** Maintain a map of event names to listener sets.

```typescript
type Listener<T> = (data: T) => void;

class EventBus<Events extends Record<string, unknown>> \{
  private listeners: \{
    [K in keyof Events]?: Set<Listener<Events[K]>>;
  \} = \{\};

  on<K extends keyof Events>(event: K, listener: Listener<Events[K]>): () => void \{
    if (!this.listeners[event]) \{
      this.listeners[event] = new Set();
    \}
    this.listeners[event]!.add(listener);

    return () => \{
      this.listeners[event]?.delete(listener);
    \};
  \}

  emit<K extends keyof Events>(event: K, data: Events[K]): void \{
    this.listeners[event]?.forEach((fn) => fn(data));
  \}
\}

interface SystemEvents \{
  userCreated: \{ id: string; name: string \};
  userDeleted: \{ id: string \};
\}

const bus = new EventBus<SystemEvents>();

const unsub = bus.on("userCreated", (data) => \{
  console.log(`created user ${data.id}`);
\});

bus.emit("userCreated", \{ id: "1", name: "Alice" \});
unsub();
```

## Exercise 3: Strategy

**Problem:**
Implement a Strategy pattern where sorting algorithms can be swapped at runtime.

**Hint:** Define a strategy interface and inject the algorithm into the context.

```typescript
interface SortStrategy<T> \{
  sort(data: T[]): T[];
\}

class BubbleSort<T> implements SortStrategy<T> \{
  sort(data: T[]): T[] \{
    const arr = [...data];
    for (let i = 0; i < arr.length; i++) \{
      for (let j = 0; j < arr.length - i - 1; j++) \{
        if (arr[j] > arr[j + 1]) \{
          [arr[j], arr[j + 1]] = [arr[j + 1], arr[j]];
        \}
      \}
    \}
    return arr;
  \}
\}

class QuickSort<T> implements SortStrategy<T> \{
  sort(data: T[]): T[] \{
    if (data.length <= 1) return [...data];
    const pivot = data[0];
    const left = data.slice(1).filter((x) => x <= pivot);
    const right = data.slice(1).filter((x) => x > pivot);
    return [...this.sort(left), pivot, ...this.sort(right)];
  \}
\}

class Sorter<T> \{
  private strategy: SortStrategy<T>;

  constructor(strategy: SortStrategy<T>) \{
    this.strategy = strategy;
  \}

  setStrategy(strategy: SortStrategy<T>): void \{
    this.strategy = strategy;
  \}

  sort(data: T[]): T[] \{
    return this.strategy.sort(data);
  \}
\}
```

## Exercise 4: Decorator

**Problem:**
Implement function decorators for logging and timing in TypeScript.

**Hint:** Use higher-order functions that wrap the original function.

```typescript
function logged<T extends (...args: any[]) => any>(fn: T): T \{
  return ((...args: any[]) => \{
    console.log(`calling ${fn.name} with`, args);
    const result = fn(...args);
    console.log(`${fn.name} returned`, result);
    return result;
  \}) as T;
\}

function timed<T extends (...args: any[]) => any>(fn: T): T \{
  return ((...args: any[]) => \{
    const start = performance.now();
    const result = fn(...args);
    const duration = performance.now() - start;
    console.log(`${fn.name} took ${duration.toFixed(2)}ms`);
    return result;
  \}) as T;
\}

function memoize<T extends (...args: any[]) => any>(fn: T): T \{
  const cache = new Map<string, any>();
  return ((...args: any[]) => \{
    const key = JSON.stringify(args);
    if (cache.has(key)) \{
      return cache.get(key);
    \}
    const result = fn(...args);
    cache.set(key, result);
    return result;
  \}) as T;
\}

const add = logged(
  timed((a: number, b: number) => a + b)
);

add(2, 3);
// calling add with [2, 3]
// add took 0.01ms
// add returned 5
```

## Exercise 5: Factory Pattern

**Problem:**
Implement a Factory pattern that creates different notification senders based on configuration.

**Hint:** Define a common interface and a factory that returns the correct implementation.

```typescript
interface NotificationSender \{
  send(to: string, message: string): Promise<boolean>;
\}

class EmailSender implements NotificationSender \{
  async send(to: string, message: string): Promise<boolean> \{
    console.log(`email to ${to}: ${message}`);
    return true;
  \}
\}

class SmsSender implements NotificationSender \{
  async send(to: string, message: string): Promise<boolean> \{
    console.log(`sms to ${to}: ${message}`);
    return true;
  \}
\}

class PushSender implements NotificationSender \{
  async send(to: string, message: string): Promise<boolean> \{
    console.log(`push to ${to}: ${message}`);
    return true;
  \}
\}

type SenderType = "email" | "sms" | "push";

class NotificationFactory \{
  private static senders: Record<SenderType, () => NotificationSender> = \{
    email: () => new EmailSender(),
    sms: () => new SmsSender(),
    push: () => new PushSender(),
  \};

  static create(type: SenderType): NotificationSender \{
    const factory = this.senders[type];
    if (!factory) \{
      throw new Error(`unknown sender type: ${type}`);
    \}
    return factory();
  \}
\}

class NotificationService \{
  async notify(type: SenderType, to: string, message: string): Promise<boolean> \{
    const sender = NotificationFactory.create(type);
    return sender.send(to, message);
  \}
\}
```

## Intuition

**TypeScript patterns solve common design problems:** Discriminated unions, mapped types, conditional types, and template literal types provide powerful abstractions that express complex constraints in the type system.

**Why it matters:** These patterns enable type-safe code that would be impossible with JavaScript alone, catching bugs at compile time.

**The key insight:** Discriminated unions (tagged unions) make impossible states unrepresentable — the compiler ensures you handle every possible case.

## Common Mistakes

**Using type assertions instead of type narrowing:** `const el = document.getElementById("app") as HTMLDivElement` assumes the element exists and is a div. If it doesnt exist, this crashes at runtime. Use instanceof or optional chaining instead.

**Ignoring the difference between `unknown` and `any`:** `unknown` is type-safe — you must narrow before using. `any` is type-unsafe — it bypasses all checking. Prefer `unknown` for values whose type is uncertain.

**Over-engineering patterns:** Design patterns add abstraction. Use them only when the complexity they solve is real. Over-engineering makes code harder to understand without providing real benefits. Start simple, refactor when patterns emerge.
